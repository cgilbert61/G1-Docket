1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.prevalidator.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "g1-docket.fullname" . }}-prevalidator)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.prevalidator.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "g1-docket.fullname" . }}-prevalidator'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "g1-docket.fullname" . }}-prevalidator --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.prevalidator.service.port }}
{{- else if contains "ClusterIP" .Values.prevalidator.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "g1-docket.name" . }},app.kubernetes.io/component=prevalidator" -o jsonpath="{.items[0].metadata.name}")
  echo "Forwarding Prevalidator Admission Gateway to http://127.0.0.1:18082"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 18082:{{ .Values.prevalidator.service.targetPort }}
{{- end }}
